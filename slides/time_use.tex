\begin{frame}
\vfill
\begin{center}
{\LARGE\bfseries How many hours a year does a researcher\\[0.3em] spend on each research task?}
\end{center}
\vfill
\end{frame}}
\addtocounter{framenumber}{-1}

% ---------------------------------------------------------------------------
\begin{frame}{Two types of tasks}
\vfill
\begin{itemize}\setlength\itemsep{1.2em}
  \item \textbf{Paper tasks} are done to produce particular papers.
  \begin{itemize}
    \item collecting a dataset, running the estimation, drafting the manuscript
  \end{itemize}
  \item \textbf{General tasks} are ongoing and not tied to particular papers.
  \begin{itemize}
    \item teaching a course, supervising students, reviewing for journals
  \end{itemize}
\end{itemize}
\vfill
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Paper tasks}
\vfill
{\large
\[
\tx{hours per year} = \HI{\tx{hours per instance}} \times \IN{\tx{instances per paper}} \times \PY{\tx{papers per year}}
\]}
\vfill
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Paper tasks: the components}
\vfill
\begin{center}
$\tx{hours per year} = \HI{\tx{hours per instance}} \times \IN{\tx{instances per paper}} \times \PY{\tx{papers per year}}$
\end{center}
\vspace{0.6em}
\small
\begin{align*}
\HI{\tx{hours per instance}} &= \tx{team hours one instance takes} \times \tx{share not counted under another task}\\[0.4em]
\IN{\tx{instances per paper}} &= \tx{share of papers that involve the task} \times
  \frac{\tx{instances per paper that involves it}}{\tx{papers per instance}}\\[0.4em]
\PY{\tx{papers per year}} &= \frac{\tx{papers a researcher publishes per year}}{\tx{authors per paper}}
\end{align*}
\vfill
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{\HI{Hours per instance}}
\begingroup\small
\[
\HI{\tx{hours per instance}} = \tx{team hours one instance takes} \times \tx{share not counted under another task}
\]
\endgroup
\vspace{-0.8em}
\divider
\small
\textbf{Team hours one instance takes}
\begin{itemize}
  \item Estimated by Claude Opus 4.8 from each task's substeps.
  \item Validated against lab protocols from protocols.io.
\end{itemize}
\vspace{0.1em}
\textbf{Share not counted under another task}
\begin{itemize}
  \item Determined by Claude Opus 5.5: it flags pairs of tasks whose steps contain the same work and counts that work
  under one of them, the keeper. The other task loses the share of its hours that is the shared work.
  \item \desc{39\% of paper tasks lose some work this way; the median one keeps 60\% of its hours.}
  \item \desc{Example: \emph{Decompose wage gaps between groups} and \emph{Decompose aggregate outcomes into components}
  both run Oaxaca--Blinder decompositions; the second keeps 30\%.}
\end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{\IN{Instances per paper}}
\begingroup\small
\[
\IN{\tx{instances per paper}} = \tx{share of papers that involve the task} \times
  \frac{\tx{instances per paper that involves it}}{\tx{papers per instance}}
\]
\endgroup
\vspace{-0.8em}
\divider
\small
\textbf{Share of papers that involve the task}
\begin{itemize}
  \item About 100 papers sampled per subfield; Claude Sonnet 5 reads each and judges whether it involves the task.
  \item \desc{Median 17\% of papers (IQR 7--41\%).}
\end{itemize}
\vspace{0.1em}
\textbf{Instances per paper that involves it} and \textbf{papers per instance}
\begin{itemize}
  \item Determined by Claude Opus 5.5 from each task's description and substeps.
  \item \desc{Instances per paper: usually 1 (69\% of tasks). \emph{Respond to peer reviewers}: 2, two rounds per
  paper.}
  \item \desc{Papers per instance: above 1 when one instance feeds several papers. \emph{Harmonize occupation and
  industry codes across data sources}: 1.5, since a harmonized crosswalk is reused in later papers. Below 1 when some
  instances never become a paper.}
\end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{\PY{Papers per year}}
\begingroup\small
\[
\PY{\tx{papers per year}} = \frac{\tx{papers a researcher publishes per year}}{\tx{authors per paper}}
\]
\endgroup
\vspace{-0.8em}
\divider
\small
\begin{itemize}\setlength\itemsep{0.6em}
  \item From OpenAlex, journal articles published 2015--2019 in each subfield.
  \item Authors per paper: the mean over papers up to the subfield's 90th percentile of authors.
  \item \desc{Medians across subfields: 2.0 papers per year and 2.6 authors per paper. Authors per paper range from 1.0
  (Religious History) to 5.8 (Cancer Biology).}
\end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{General tasks}
\vfill
\begin{center}
$\tx{hours per year} = \HI{\tx{hours per instance}} \times \IY{\tx{instances per year}}$
\end{center}
\vspace{0.6em}
\small
\begin{align*}
\HI{\tx{hours per instance}} &= \tx{hours the researcher spends per instance}\\
&\quad\times \tx{share not counted under another task}\\[0.4em]
\IY{\tx{instances per year}} &= \tx{share of researchers who do it} \times \tx{instances per year for each of them}
\end{align*}
\vspace{0.4em}
\begin{itemize}
  \item Hours per instance for seven ongoing tasks shared by all fields (supervising, teaching a course, keeping up,
  learning methods, reviewing manuscripts, commenting on theses, editorial and committee roles) are the median across
  fields.
\end{itemize}
\vfill
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{\IY{Instances per year}}
\begingroup\small
\[
\IY{\tx{instances per year}} = \tx{share of researchers who do it} \times \tx{instances per year for each of them}
\]
\endgroup
\vspace{-0.8em}
\divider
\small
\textbf{Share of researchers who do it}
\begin{itemize}
  \item Estimated by Claude Opus 5.
  \item \desc{Median 90\% for general tasks, as for subfield-level tasks.}
\end{itemize}
\vspace{0.1em}
\textbf{Instances per year for each of them}
\begin{itemize}
  \item Determined by Claude Sonnet 5.5, with the task's substeps shown.
  \item \desc{Median 1.5 a year. \emph{Evaluate manuscripts for journals} in Labor Economics: 95\% review, 6 manuscripts
  each, so $95\% \times 6 = 5.7$ reviews per researcher per year.}
\end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Worked example: a paper task in Labor Economics}
\textbf{Respond to peer reviewer comments and revise manuscripts for publication}\\[0.6em]
\small
One round of responses and revisions takes the team 121 hours, none of it counted under another task. 72\% of labor
economics papers go through revision, with two rounds each. A labor economist publishes 2.0 papers a year with 1.76
authors per paper.
\begin{align*}
\HI{\tx{hours per instance}} &= 121 \times 100\% = \HI{121}\\
\IN{\tx{instances per paper}} &= 72\% \times 2 \div 1 = \IN{1.44}\\
\PY{\tx{papers per year}} &= 2.0 \div 1.76 = \PY{1.14}\\[0.4em]
\tx{hours per year} &= \HI{121} \times \IN{1.44} \times \PY{1.14} \approx \textbf{198}
\end{align*}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Worked example: a general task in Labor Economics}
\textbf{Supervise graduate students and postdoctoral researchers on research projects}\\[0.6em]
\small
Supervising one student for a year takes 88.5 hours, the median across fields, none of it counted under another task. 89\% of labor economists
supervise, three students each.
\begin{align*}
\HI{\tx{hours per instance}} &= 88.5 \times 100\% = \HI{88.5}\\
\IY{\tx{instances per year}} &= 89\% \times 3 = \IY{2.7}\\[0.4em]
\tx{hours per year} &= \HI{88.5} \times \IY{2.7} \approx \textbf{236}
\end{align*}
\end{frame}
